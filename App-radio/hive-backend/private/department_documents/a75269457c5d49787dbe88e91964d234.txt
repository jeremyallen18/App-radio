# ============================================================
# ANALISIS INTELIGENTE DE DATOS
# ANALISIS DE BASES DE DATOS PARA CLASIFICACION
# ============================================================

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import os
from google.colab import files
from IPython.display import display

# ============================================================
# 1. CARGAR BASES DE DATOS
# ============================================================

archivos_subidos = files.upload()


# ============================================================
# 2. FUNCION PARA CARGAR LOS ARCHIVOS
# ============================================================

def cargar_datos(nombre_archivo):

    extension = os.path.splitext(nombre_archivo)[1].lower()

    if extension in [".xlsx", ".xls"]:
        datos = pd.read_excel(nombre_archivo)

    elif extension == ".csv":

        try:
            datos = pd.read_csv(nombre_archivo)
        except:
            datos = pd.read_csv(nombre_archivo, sep=";")

    else:
        return None

    # Eliminar solamente filas y columnas
    # completamente vacias
    datos = datos.dropna(axis=0, how="all")
    datos = datos.dropna(axis=1, how="all")

    return datos


# ============================================================
# 3. DETECTAR AUTOMATICAMENTE EL ATRIBUTO CLASE
# ============================================================

def detectar_clase(datos):

    posibles_clases = [
        "class",
        "clase",
        "target",
        "label",
        "category",
        "categoria",
        "survived",
        "class real"
    ]

    for columna in datos.columns:

        nombre = str(columna).strip().lower()

        if nombre in posibles_clases:
            return columna

    # Si no encuentra un nombre conocido,
    # utiliza la ultima columna como clase
    return datos.columns[-1]


# ============================================================
# 4. ANALIZAR CADA BASE DE DATOS
# ============================================================

for nombre_archivo in archivos_subidos:

    datos = cargar_datos(nombre_archivo)

    if datos is None:
        print("Archivo no compatible:", nombre_archivo)
        continue

    # --------------------------------------------------------
    # IDENTIFICAR CLASE
    # --------------------------------------------------------

    columna_clase = detectar_clase(datos)

    # Eliminar solamente registros sin clase
    datos = datos.dropna(subset=[columna_clase])

    # --------------------------------------------------------
    # CARACTERISTICAS
    # --------------------------------------------------------

    caracteristicas = [
        columna
        for columna in datos.columns
        if columna != columna_clase
    ]

    numero_caracteristicas = len(caracteristicas)

    # --------------------------------------------------------
    # CLASES
    # --------------------------------------------------------

    clases = datos[columna_clase].dropna().unique()

    numero_clases = len(clases)

    # ========================================================
    # 1. NUMERO DE CLASES
    # ========================================================

    print("Numero de clases:", numero_clases)

    # ========================================================
    # 2. NUMERO DE CARACTERISTICAS
    # ========================================================

    print("Numero de características:", numero_caracteristicas)

    # ========================================================
    # 3. GRAFICA DE BARRAS DE LAS CLASES
    # ========================================================

    cantidad_clases = (
        datos[columna_clase]
        .value_counts()
        .sort_index()
    )

    plt.figure(figsize=(8, 5))

    plt.bar(
        cantidad_clases.index.astype(str),
        cantidad_clases.values
    )

    plt.xlabel("Clase")
    plt.ylabel("Numero de patrones")
    plt.title(
        "Distribucion de clases - " + nombre_archivo
    )

    plt.tight_layout()
    plt.show()


    # ========================================================
    # 4. MAPA DE DISPERSION DE LAS CLASES
    # ========================================================

    # Buscar TODOS los atributos numericos
    atributos_numericos = datos[
        caracteristicas
    ].select_dtypes(
        include=np.number
    ).columns.tolist()

    if len(atributos_numericos) >= 2:

        # ----------------------------------------------------
        # Seleccionar las dos primeras características
        # numéricas
        # ----------------------------------------------------

        caracteristica_x = atributos_numericos[0]
        caracteristica_y = atributos_numericos[1]

        # ----------------------------------------------------
        # Crear copia exclusivamente para la gráfica
        # ----------------------------------------------------

        datos_grafica = datos[
            [caracteristica_x,
             caracteristica_y,
             columna_clase]
        ].copy()

        # Eliminar SOLO registros que no puedan ubicarse
        # en el plano porque tienen un valor vacío
        datos_grafica = datos_grafica.dropna(
            subset=[
                caracteristica_x,
                caracteristica_y
            ]
        )

        plt.figure(figsize=(10, 8))

        # ----------------------------------------------------
        # Dibujar CADA REGISTRO de cada clase
        # ----------------------------------------------------

        for clase in clases:

            datos_clase = datos_grafica[
                datos_grafica[columna_clase] == clase
            ]

            plt.scatter(
                datos_clase[caracteristica_x],
                datos_clase[caracteristica_y],
                label=f"Clase {clase}",
                alpha=0.75,
                s=45
            )

        # ----------------------------------------------------
        # Configuracion de la grafica
        # ----------------------------------------------------

        plt.xlabel(caracteristica_x)
        plt.ylabel(caracteristica_y)

        plt.title(
            "Mapa de dispersion de las clases - "
            + nombre_archivo
        )

        plt.legend(title="Clases")

        plt.grid(
            True,
            alpha=0.3
        )

        plt.tight_layout()
        plt.show()

        # ----------------------------------------------------
        # COMPROBAR CUANTOS DATOS SE MOSTRARON
        # ----------------------------------------------------

        print(
            "Registros de la base de datos:",
            len(datos)
        )

        print(
            "Registros representados en el mapa:",
            len(datos_grafica)
        )

        registros_no_graficados = (
            len(datos) - len(datos_grafica)
        )

        print(
            "Registros sin representar por valores "
            "faltantes en las características seleccionadas:",
            registros_no_graficados
        )

    else:

        print(
            "No existen al menos dos características "
            "numéricas para generar el mapa de dispersion."
        )


    # ========================================================
    # TABLA
    # ========================================================

    numero_instancias = len(datos)


    # --------------------------------------------------------
    # NUMERO DE PATRONES POR CLASE
    # --------------------------------------------------------

    patrones_por_clase = (
        datos[columna_clase]
        .value_counts()
        .sort_index()
    )

    patrones_texto = ""

    for clase, cantidad in patrones_por_clase.items():

        patrones_texto += (
            f"Clase {clase}: {cantidad} patrones\n"
        )


    # --------------------------------------------------------
    # CARACTERISTICAS DE LOS ATRIBUTOS
    # --------------------------------------------------------

    caracteristicas_atributos = ""

    for atributo in caracteristicas:

        tipo = datos[atributo].dtype

        valores_unicos = datos[atributo].nunique(
            dropna=True
        )

        valores_nulos = datos[atributo].isna().sum()

        if pd.api.types.is_numeric_dtype(
            datos[atributo]
        ):

            minimo = datos[atributo].min()
            maximo = datos[atributo].max()
            media = datos[atributo].mean()

            caracteristicas_atributos += (
                f"{atributo}: Numerico | "
                f"min={minimo} | "
                f"max={maximo} | "
                f"media={media:.2f} | "
                f"valores unicos={valores_unicos} | "
                f"valores vacios={valores_nulos}\n"
            )

        else:

            caracteristicas_atributos += (
                f"{atributo}: Categorico | "
                f"valores unicos={valores_unicos} | "
                f"valores vacios={valores_nulos}\n"
            )


    # --------------------------------------------------------
    # FECHA DE LOS DATOS DONADOS
    # --------------------------------------------------------

    fecha_donacion = "No especificada"


    # ========================================================
    # CREAR UNA SOLA TABLA
    # ========================================================

    tabla = pd.DataFrame({

        "Información": [

            "Numero de instancias",
            "Numero de atributos",
            "Numero de patrones en cada una de las clases de la base de datos",
            "Características de los atributos",
            "Fecha de los datos donados",
            "Tarea asociada"

        ],

        "Resultado": [

            numero_instancias,

            numero_caracteristicas,

            patrones_texto,

            caracteristicas_atributos,

            fecha_donacion,

            "Clasificación"

        ]

    })


    # ========================================================
    # CONFIGURACION PARA MOSTRAR TODA LA INFORMACION
    # ========================================================

    pd.set_option(
        "display.max_colwidth",
        None
    )

    pd.set_option(
        "display.max_rows",
        None
    )

    pd.set_option(
        "display.max_columns",
        None
    )

    pd.set_option(
        "display.width",
        None
    )


    # ========================================================
    # MOSTRAR TABLA
    # ========================================================

    display(tabla)

    print("\n")
